#----->>>
with repeat_cust as (
select c1.order_date,count(distinct c1.customer_id) as repeat_cust from customers c1, customers c2 
where c1.customer_id=c2.customer_id and c1.order_date>c2.order_date
group by c1.order_date ),

total_cust as (select order_date,count(1) as tot_cust from customers group by order_date)

select total_cust.order_date as order_date,tot_cust-coalesce(repeat_cust,0) as new_customers,coalesce(repeat_cust,0) as repaet_customers
 from total_cust left join repeat_cust on total_cust.order_date=repeat_cust.order_date;
#--------->
# Another way
SELECT order_date,
    Count(CASE WHEN rnk = 1 THEN cnt END) AS "New Customer",
    Count(CASE WHEN rnk > 1 THEN cnt END) AS "Old Customer"
FROM (
    SELECT order_date, customer_id, 
        DENSE_RANK() OVER (PARTITION BY customer_id ORDER BY order_date) AS rnk,
        COUNT(*) OVER (PARTITION BY customer_id, order_date) AS cnt
    FROM customers
) A
GROUP BY order_date;
# --------> another way
Select a.order_date,
Sum(Case when a.order_date = a.first_order_date then 1 else 0 end) as new_customer,
Sum(Case when a.order_date != a.first_order_date then 1 else 0 end) as repeat_customer
from(
Select customer_id, order_date, min(order_date) over(partition by customer_id) as first_order_date from customers) a 
group by a.order_date;

#--------------> another way
with cte as (
select *,
    count(order_id) over(partition by customer_id 
                        order by order_date 
                        rows between unbounded preceding and current row) as cnt
from customers
)
select order_date, 
    sum(case when cnt = 1 then 1 else 0 end) as new_cust_ind,
    sum(case when cnt > 1 then 1 else 0 end) as repeat_cust_ind,
    sum(case when cnt = 1 then order_amount else 0 end) as new_cust_amt,
    sum(case when cnt > 1 then order_amount else 0 end) as repeat_cust_amt
from cte
group by order_date;

create table entries ( 
name varchar(20),
address varchar(20),
email varchar(20),
floor int,
resources varchar(10));

insert into entries 
values ('A','Bangalore','A@gmail.com',1,'CPU'),('A','Bangalore','A1@gmail.com',1,'CPU'),('A','Bangalore','A2@gmail.com',2,'DESKTOP')
,('B','Bangalore','B@gmail.com',2,'DESKTOP'),('B','Bangalore','B1@gmail.com',2,'DESKTOP'),('B','Bangalore','B2@gmail.com',1,'MONITOR');

select * from entries;

with ct as (select name,floor,count(1) as cnt from entries group by name,floor),
ct2 as (select c1.* from ct c1 join ct c2 on c1.name=c2.name and c1.cnt>c2.cnt),  
main as (select name,count(1) as total_visits,group_concat(distinct resources) as resources_used from entries group by name)
select a.name,a.total_visits,b.floor as most_visited_floor,a.resources_used from main a join ct2 b on a.name=b.name;

# another way
# Instead of using line 173 and 174 we can use below single cte as just select where rnk=1 while using this cte
with scte as (select name,floor,count(1) as cnt, rank() over (partition by name order by count(1) desc) as rnk from entries group by name,floor),
main as (select name,count(1) as total_visits,group_concat(distinct resources) as resources_used from entries group by name)
select a.name,a.total_visits,b.floor as most_visited_floor,a.resources_used from main a join scte b on a.name=b.name where rnk=1;

### Pareto Principle (80/20 Rule) Implementation in SQL ##################################################################################

select * from superstore_orders;    
select .8*sum(sales) from superstore_orders; #21165.489360000003
with prod_wise_sales as (
select product_name,sum(sales) as prod_wise_sales from superstore_orders group by product_name),
final_cte as (
select *,sum(prod_wise_sales) over (order by prod_wise_sales desc rows between unbounded preceding and 0 preceding) as running_sales,.8*sum(prod_wise_sales) over () as total_sales from prod_wise_sales)
select * from final_cte where running_sales<total_sales

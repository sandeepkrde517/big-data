# Moving avg
# In a DataFrame df_sales with columns Date, ProductID, and QuantitySold, how would you calculate a 7-day rolling average of QuantitySold for each product?
# Sample data
from pyspark.sql import Row
from pyspark.sql.window import Window
from pyspark.sql import functions as f

data = [Row(Date='2023-01-01', ProductID=100, QuantitySold=10),
        Row(Date='2023-01-02', ProductID=100, QuantitySold=15),
        Row(Date='2023-01-03', ProductID=100, QuantitySold=20),
        Row(Date='2023-01-04', ProductID=100, QuantitySold=25),
        Row(Date='2023-01-05', ProductID=100, QuantitySold=30),
        Row(Date='2023-01-06', ProductID=100, QuantitySold=35),
        Row(Date='2023-01-07', ProductID=100, QuantitySold=40),
        Row(Date='2023-01-08', ProductID=100, QuantitySold=45)]

# Create DataFrame
df_sales = spark.createDataFrame(data)
# df_sales.show()
# df_sales.printSchema()
df_sales_date = df_sales.withColumn('Date',f.to_date(df_sales['Date']))
# df_sales_date.show()
# df_sales_date.printSchema()
# Window specification for 7-day rolling average
windowSpec = Window.partitionBy('ProductID').orderBy('Date').rowsBetween(-6, 0)

df_sales_moving_avg = df_sales_date.withColumn('7DaysMovingAvg',f.avg('QuantitySold').over(windowSpec))

df_sales_moving_avg.show()

# How can you use UDFs (User Defined Functions) in PySpark to apply a complex transformation, say, categorizing ages into groups ('Youth', 'Adult', 'Senior')?

from pyspark.sql import Row
from pyspark.sql.functions import udf
from pyspark.sql.types import StringType

# Sample data
data = [Row(UserID=4001, Age=17),
        Row(UserID=4002, Age=45),
        Row(UserID=4003, Age=65),
        Row(UserID=4004, Age=30),
        Row(UserID=4005, Age=80)]

# Create DataFrame
df = spark.createDataFrame(data)

# Define UDF to categorize age
def categorize_age(age):
    if age < 18:
        return 'Youth'
    elif age < 60:
        return 'Adult'
    else:
        return 'Senior'
    
# age_udf = udf(categorize_age, StringType())
age_udf = udf(categorize_age, 'string')

# Apply UDF to categorize ages
df = df.withColumn('AgeGroup', age_udf(df['Age']))

# Show results
df.show()

# SCD Type 2 implementation using pyspark
## As part of SCD 2 we have to maintain history 
# We can do this by mainly 5 steps
## 1. collect the updated records with updated information
## 2. collect the older records by updating their required columns 
## 3. collect the new records 
## 4. Merge all (target + step1 + step2)
## 5. Remove duplicates which came due to joins

from pyspark.sql import functions as f
from pyspark.sql.window import Window as w
### Input datasets

customer_dim_data = [
(1,'manish','arwal','india','N','2022-09-15','2022-09-25'),
(2,'vikash','patna','india','Y','2023-08-12',None),
(3,'nikita','delhi','india','Y','2023-09-10',None),
(4,'rakesh','jaipur','india','Y','2023-06-10',None),
(5,'ayush','NY','USA','Y','2023-06-10',None),
(1,'manish','gurgaon','india','Y','2022-09-25',None),
]
customer_schema= ['id','name','city','country','active','effective_start_date','effective_end_date']
customer_dim_df = spark.createDataFrame(data= customer_dim_data,schema=customer_schema)

sales_data = [
(1,1,'manish','2023-01-16','gurgaon','india',380),
(77,1,'manish','2023-03-11','bangalore','india',300),
(12,3,'nikita','2023-09-20','delhi','india',127),
(54,4,'rakesh','2023-08-10','jaipur','india',321),
(65,5,'ayush','2023-09-07','mosco','russia',765),
(89,6,'rajat','2023-08-10','jaipur','india',321)
]
sales_schema = ['sales_id', 'customer_id','customer_name', 'sales_date', 'food_delivery_address','food_delivery_country', 'food_cost']
sales_df = spark.createDataFrame(data=sales_data,schema=sales_schema)

# customer_dim_df.show()
# sales_df.show()

### Step 1 >> updated records with new information

joined_df = customer_dim_df.join(sales_df,(customer_dim_df['id']==sales_df['customer_id']),'left')
# joined_df.show()
joined_df_filtered = joined_df.filter((f.col('city')!=f.col('food_delivery_address')) & (f.col('active')=='Y'))
# joined_df_filtered.cache()
# joined_df_filtered.show()
joined_df_updated = joined_df_filtered.select('customer_id','customer_name',f.col('food_delivery_address').alias('city'),'food_delivery_country','active','effective_start_date','effective_end_date')
joined_df_updated.show()

### step 2 >> old records with updated information

old_records_updated_df = joined_df_filtered.withColumn('effective_end_date',f.col('sales_date')).withColumn('active',f.lit('N')).select('customer_id','customer_name','city','food_delivery_country','active','effective_start_date','effective_end_date')
old_records_updated_df.show()

### Step 3 >> new records
new_records_df = sales_df.join(customer_dim_df,(customer_dim_df['id']==sales_df['customer_id']),'left_anti')
# new_records_df.show()
new_records_add_cols_df = new_records_df.withColumn('effective_start_date',f.col('sales_date')).withColumn('effective_end_date',f.lit(None)).withColumn('active',f.lit('Y')).select('customer_id','customer_name','food_delivery_address','food_delivery_country','active','effective_start_date','effective_end_date')
new_records_add_cols_df.show()

### Step 4 >> Merge all
merged_df = customer_dim_df.union(joined_df_updated).union(old_records_updated_df).union(new_records_add_cols_df).orderBy('id','active','effective_end_date')
merged_df.show()

### Step 5 >> Remove duplicate records
final_df = merged_df.withColumn('rnk',f.rank().over(w.partitionBy('id','city').orderBy(f.col('effective_end_date').desc()))).filter("rnk==1").drop('rnk').orderBy('id','active','effective_end_date')
customer_dim_df.orderBy('id').show()
sales_df.orderBy('customer_id').show()
final_df.show()
